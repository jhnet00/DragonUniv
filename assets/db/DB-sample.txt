CREATE TABLE students (
    stu_id INT PRIMARY KEY AUTO_INCREMENT,
    stu_name VARCHAR(50) NOT NULL,
    stu_number VARCHAR(20) UNIQUE NOT NULL,
    major_id INT NOT NULL,
    FOREIGN KEY (major_id) REFERENCES majors(major_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);


CREATE TABLE stu_classes (
    id INT PRIMARY KEY AUTO_INCREMENT,
    stu_id INT NOT NULL,
    class_id INT NOT NULL,
    FOREIGN KEY (stu_id) REFERENCES students(stu_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (class_id) REFERENCES classes(class_id)
        ON DELETE CASCADE ON UPDATE CASCADE
);

// Use DBML to define your database structure

Table majors {
  major_id int [primary key]
  Major_name varchar(50) [not null]

  Note: 'majors table'
}

Table classes {
  class_id int [pk, increment]
  class varchar(50) [not null]
  major varchar(50) [not null]
  class_Name varchar(50) [not null]

  Note: 'classes table'
}

Table students {
  stu_id int [pk, increment]
  stu_name varchar(50) [not null]
  stu_number varchar(20) [unique, not null]
  password varchar(60) [not null]
  major_id int [not null]

  Note: 'students table'
}

Table stu_classes {
  id int [pk, increment]
  stu_id int [not null]
  class_id int [not null]
}

Table admins {
  id int [pk, increment]
  admin_user varchar(50) [unique, not null]
  admin_pass varchar(60) [not null]
  major_id int [not null]

  Note: 'admins table'
}

Ref: stu_classes.stu_id > students.stu_id [delete: cascade, update: cascade]
Ref: stu_classes.class_id > classes.class_id [delete: cascade, update: cascade]
Ref: students.major_id > majors.major_id [delete: cascade, update: cascade]
